```
The directory `docs` contains all the files related to the ontology, its versions in time, and the related documentations. In particular, it includes:

* the `current` directory, where the files of the current version of the ontology are stored;
* one `yyyy-mm-dd` version directory for each of the versions of the ontology developed.

The `current` directory contains five files, named after the lowercase ontology acronym, with the following extensions specifying different formats: `.xml` (RDF/XML), `.ttl` (Turtle), `.nt` (Ntriple), `.jsonld` (JSON-LD), `.html` (HTML, i.e. the human readable documentation of the ontology). All the images used in the documentation should additionally be included in this directory.

The version directories (i.e. `yyyy-mm-dd`) contains the same kinds of files as those included in the `current` directory, but specific for that particular version.
```

```
import argparse
from datetime import datetime
import subprocess
import sys
import tomllib
from pathlib import Path
import requests
from owlrl import OWLRL_Semantics
from rdflib import Graph

ROOT = Path(__file__).resolve().parent

_pyproject = tomllib.loads((ROOT / "pyproject.toml").read_text(encoding="utf-8"))
ACRONYM = _pyproject["project"]["name"]
_config = _pyproject.get("tool", {}).get("validate", {})
ONTOLOGY_FILE = ROOT / _config.get("ontology_file", f"docs/current/{ACRONYM}.ttl")
NAMESPACE = _config.get("namespace", f"http://purl.org/spar/{ACRONYM}/")
FOOPS_API_URL = _config.get("foops_api_url", "https://foops.linkeddata.es/assessOntologyFile")
MIN_FAIR_SCORE = _config.get("min_fair_score", 0.85)

PREFIXES = """
PREFIX owl: <http://www.w3.org/2002/07/owl#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
"""

DL_CHECKS = {
    "IRI declared as both object and datatype property": """
        SELECT DISTINCT ?x WHERE { ?x a owl:ObjectProperty, owl:DatatypeProperty . }
    """,
    "IRI declared as both object and annotation property": """
        SELECT DISTINCT ?x WHERE { ?x a owl:ObjectProperty, owl:AnnotationProperty . }
    """,
    "IRI declared as both datatype and annotation property": """
        SELECT DISTINCT ?x WHERE { ?x a owl:DatatypeProperty, owl:AnnotationProperty . }
    """,
    "Transitive property used in a cardinality restriction": """
        SELECT DISTINCT ?x WHERE {
            ?x a owl:TransitiveProperty .
            ?r owl:onProperty ?x .
            { ?r owl:cardinality ?n } UNION { ?r owl:minCardinality ?n }
            UNION { ?r owl:maxCardinality ?n } UNION { ?r owl:qualifiedCardinality ?n }
            UNION { ?r owl:minQualifiedCardinality ?n } UNION { ?r owl:maxQualifiedCardinality ?n }
        }
    """,
    "Datatype property declared transitive, symmetric or inverse": """
        SELECT DISTINCT ?x WHERE {
            ?x a owl:DatatypeProperty .
            { ?x a owl:TransitiveProperty } UNION { ?x a owl:SymmetricProperty }
            UNION { ?x owl:inverseOf ?y } UNION { ?y owl:inverseOf ?x }
        }
    """,
    "Ontology term used but not declared": f"""
        SELECT DISTINCT ?x WHERE {{
            {{ ?s ?x ?o }} UNION {{ ?s rdf:type ?x }} UNION {{ ?s rdfs:subClassOf ?x }}
            UNION {{ ?s rdfs:domain ?x }} UNION {{ ?s rdfs:range ?x }}
            UNION {{ ?s owl:onProperty ?x }} UNION {{ ?s owl:someValuesFrom ?x }}
            UNION {{ ?s owl:allValuesFrom ?x }} UNION {{ ?s rdfs:subPropertyOf ?x }}
            FILTER(isIRI(?x) && STRSTARTS(STR(?x), "{NAMESPACE}"))
            FILTER NOT EXISTS {{ ?x a ?t }}
        }}
    """,
}


REPORT_FILE = ROOT / "validation_report.txt"
report = []
failures = []
warnings = []


def log(message):
    print(message)
    report.append(message)


def fail(message):
    log(f"✗ {message}")
    failures.append(message)


def warn(message):
    log(f"⚠ {message}")
    warnings.append(message)


def finish():
    status = f"FAILED ({len(failures)} check(s))" if failures else "PASSED"
    if warnings:
        status += f" with {len(warnings)} warning(s)"
    log(f"\nResult: {status}")
    header = f"{ACRONYM.upper()} validation report\nFile: {ONTOLOGY_FILE.relative_to(ROOT)}\nDate: {datetime.now():%Y-%m-%d %H:%M:%S}\n"
    REPORT_FILE.write_text(header + "\n" + "\n".join(report) + "\n", encoding="utf-8")
    print(f"Report written to {REPORT_FILE.relative_to(ROOT)}")
    sys.exit(1 if failures else 0)


def check_syntax():
    log(f"-> Checking RDF syntax ({ONTOLOGY_FILE.relative_to(ROOT)})...")
    if not ONTOLOGY_FILE.exists():
        fail(f"File not found: {ONTOLOGY_FILE}")
        finish()
    graph = Graph()
    try:
        graph.parse(ONTOLOGY_FILE, format="turtle")
    except SyntaxError as e:
        fail(f"RDF parsing error: {e}")
        finish()
    log("✓ RDF Turtle syntax is valid.")
    return graph


def check_spelling():
    log("-> Checking spelling with codespell (warnings only)...")
    result = subprocess.run(["codespell", str(ONTOLOGY_FILE)], capture_output=True, text=True)
    if result.returncode != 0:
        log(result.stdout.replace(str(ROOT) + "/", "").rstrip())
        warn("Possible spelling errors found: check them manually, they may be correct terms.")
        return
    log("✓ No spelling errors found.")


def check_dl(graph):
    log("-> Checking common OWL 2 DL violations...")
    errors = []
    for title, query in DL_CHECKS.items():
        for row in graph.query(PREFIXES + query):
            errors.append(f"{title}: {row.x}")
    if errors:
        for error in errors:
            log(f"  - {error}")
        fail("OWL 2 DL violations found.")
        return
    log("✓ No common OWL 2 DL violations found.")


def check_consistency(graph):
    log("-> Checking consistency (OWL 2 RL, imports not loaded)...")
    closure = Graph()
    for triple in graph:
        closure.add(triple)
    semantics = OWLRL_Semantics(closure, False, False, False)
    semantics.closure()
    if semantics.error_messages:
        for message in semantics.error_messages:
            log(f"  - {message}")
        fail("Inconsistencies found.")
        return
    log("✓ No inconsistencies found.")


def check_fairness():
    log("-> Running FAIR assessment via FOOPS! API...")
    try:
        with open(ONTOLOGY_FILE, "rb") as f:
            response = requests.post(
                FOOPS_API_URL,
                files={"file": (ONTOLOGY_FILE.name, f, "text/turtle")},
                headers={"Accept": "application/json"},
                timeout=60,
            )
        response.raise_for_status()
        data = response.json()
    except requests.exceptions.RequestException as e:
        if e.response is not None:
            log(f"Server response: {e.response.text}")
        fail(f"FOOPS! API request failed: {e}. Use --skip-fair to run the local checks only.")
        return
    except ValueError:
        fail(f"FOOPS! API returned a non-JSON response: {response.text[:500]}")
        return

    if "overall_score" not in data:
        fail(f"Unexpected FOOPS! response format: {str(data)[:500]}")
        return
    overall_score = data["overall_score"]

    log("\n--- FOOPS! FAIR Assessment Report ---")
    log(f"Overall FAIR Score: {overall_score * 100:.1f}%\n")

    failed_checks = []
    for check in data.get("checks", []):
        title = check.get("title", check.get("id", "Check"))
        category = check.get("principle_id", "FAIR")
        if check.get("total_passed_tests", 0) == check.get("total_tests_run", -1):
            log(f"  [✓ PASS] [{category}] {title}")
        else:
            log(f"  [✗ FAIL] [{category}] {title}")
            failed_checks.append(f"[{category}] {title}: {check.get('explanation', '')}")

    log("-------------------------------------")

    if overall_score < MIN_FAIR_SCORE:
        log("\nFailed Checks:")
        for check in failed_checks:
            log(f"  - {check}")
        fail(
            f"Quality Gate Failed: FAIR score ({overall_score * 100:.1f}%) is below "
            f"the minimum threshold ({MIN_FAIR_SCORE * 100:.1f}%)."
        )
        return

    log(f"\n✓ FAIR assessment passed ({overall_score * 100:.1f}%).")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=f"Validate the {ACRONYM.upper()} ontology.")
    parser.add_argument("--skip-fair", action="store_true", help="skip the online FOOPS! assessment")
    args = parser.parse_args()

    graph = check_syntax()
    check_spelling()
    check_dl(graph)
    check_consistency(graph)
    if not args.skip_fair:
        check_fairness()
    finish()
```

### FRAPO
* Ruoli allineati a SCoRO;
* Aggiunto il pattern di `pso:StatusInTime`;
    * `frapo:Status` diventa subclass di `pso:Status`
* Aggiunte nuove proprietà:
    * * `frapo:hasBudgetInformation` e `frapo:isBudgetInformationOf`:  domain `frapo:Budget`, range `frapo:BudgetInformation`, e inversa;
    * `frapo:hasBudgetCategory`:  domain `frapo:BudgetInformation`, range `frapo:BudgetCategory`.
    * Spiegazione: `frapo:Budget` è la classe hub, `frapo:BudgetInformation` è la singola figura nel `frapo:Budget`, e la `frapo:BudgetCategory` è l'etichetta che classifica per cosa sta la singola `frapo:BudgetInformation`.

Esempio:

    :my-budget a frapo:Budget ;
        frapo:hasBudgetInformation :travel-budgeted .
    
    :travel-budgeted a frapo:BudgetInformation ;
        frapo:hasCurrency "euro" ;
        frapo:hasMonetaryValue "1000"^^xsd:decimal ;
        frapo:hasBudgetCategory frapo:travel-and-subsistence-costs .
